(async function() {
    let allData = [];
    
    for (let i = 1; i <= 15; i++) {
        // 현재 페이지의 단어 카드나 텍스트 요소들을 지정 (실제 단어들이 담긴 HTML 태그 클래스명 확인 필요)
        // 일단 현재 화면에 있는 텍스트나 카드 안의 텍스트를 모두 수집하도록 구성합니다.
        const wordElements = document.querySelectorAll('.word-item, .word-card, tr, li'); // 구조에 맞게 수정 가능
        
        // 만약 요소 선택이 확실하지 않다면 현재 페이지의 의미 있는 텍스트 단위를 수집
        const pageTexts = Array.from(document.querySelectorAll('div, span')).map(el => el.innerText).filter(t => t.length > 0);
        
        console.log(`${i}페이지 처리 중... (수집된 항목 수: ${pageTexts.length})`);
        allData.push({ page: i, text: document.body.innerText }); // 일단 전체 텍스트를 페이지별로 담음

        // 다음 페이지 버튼 찾기 (사이트에 맞는 버튼 클래스나 화살표 요소를 찾아야 함)
        // 보통 '다음'이나 '>' 모양의 버튼입니다.
        const nextBtn = document.querySelector('.btn-next, [aria-label="Next page"], .next'); 
        
        if (nextBtn && !nextBtn.classList.contains('is-disabled') && !nextBtn.disabled) {
            nextBtn.click();
            await new Promise(resolve => setTimeout(resolve, 1500)); // 로딩 대기 시간 1.5초
        } else {
            console.log("다음 페이지 버튼을 찾지 못했거나 마지막 페이지입니다.");
            break;
        }
    }

    console.log("전체 수집 완료!", allData);

    // 결과를 파일로 자동 다운로드
    const blob = new Blob([JSON.stringify(allData, null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = 'hsk_words.json';
    document.body.appendChild(a);
    a.click();
    a.remove();
    console.log("hsk_words.json 파일 다운로드 완료!");
})();
